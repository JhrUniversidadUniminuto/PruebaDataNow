        ------------------------------        01. Registra Sql server que es el motor de base de datos elegido para la base de datos transaccional          ------------------------------

az provider register --namespace Microsoft.Sql

        -------------------------------    02. Crear Servidor (por restriccion de la cueta se crea en esa locacion centralus el servidor )   ------------------------------------------

az sql server create --resource-group PruebasDataNow `
    --name logitrackjulian2026PruebaDataNow `
    --location centralus `
    --admin-user rootadmin `
    --admin-password "admin123*"

        -------------------------------------------------------    03. Crear Base de datos Transaccional      ---------------------------------------------------------------------

az sql db create `
    --resource-group PruebasDataNow `
    --server logitrackjulian2026PruebaDataNow `
    --name LogiTrack_Transaccional `
    --service-objective Basic


    -------------------------------   (para crear la tablas debe primer configurar el firewall en el azure agregando la ip v4 para poder ejecutar este sql desde el host donde va a ejecutar las sentecias) ----------------------------------

az sql server firewall-rule create `
  --resource-group PruebasDataNow `
  --server logitrackjulian2026pruebadatanow `
  --name HostAdminBdTransaccional `
  --start-ip-address 181.51.32.95 `
  --end-ip-address 181.51.32.95

      ---------------------------------------------------        03_01. Crear Objeto de datos OPE_CONDUCTORES     -----------------------------------------------------------------

sqlcmd -S logitrackjulian2026pruebadatanow.database.windows.net `
       -d LogiTrack_Transaccional `
       -U rootadmin `
       -P "admin123*" `
       -Q "CREATE TABLE OPE_CONDUCTORES (cond_id INT IDENTITY(1,1) PRIMARY KEY, nomb_cond VARCHAR(100) NOT NULL, apell_cond VARCHAR(100) NOT NULL, tip_doc CHAR(3) NOT NULL, num_doc_hash VARCHAR(255) NOT NULL, fec_ingreso DATE NOT NULL, id_ciudad_base INT NOT NULL, tip_vehiculo VARCHAR(20) NOT NULL, cod_zona_asignada INT NOT NULL, activo BIT NOT NULL DEFAULT 1, calific_promedio_acum DECIMAL(3,2));"

      ---------------------------------------------------        03_02. Crear Objeto de datos CLI_REMITENTES     -----------------------------------------------------------------

    ---------------------------------------------------        Mostrar Fk Creadas     -----------------------------------------------------------------


sqlcmd -S logitrackjulian2026pruebadatanow.database.windows.net `
       -d LogiTrack_Transaccional `
       -U rootadmin `
       -P "admin123*" `
       -Q "SELECT fk.name AS ForeignKey,tp.name AS TablaPadre,cp.name AS ColumnaPadre,tr.name AS TablaReferenciada,cr.name AS ColumnaReferenciada FROM sys.foreign_keys fk INNER JOIN sys.foreign_key_columns fkc ON fk.object_id = fkc.constraint_object_id INNER JOIN sys.tables tp ON fkc.parent_object_id = tp.object_id INNER JOIN sys.columns cp ON fkc.parent_object_id = cp.object_id AND fkc.parent_column_id = cp.column_id INNER JOIN sys.tables tr ON fkc.referenced_object_id = tr.object_id INNER JOIN sys.columns cr ON fkc.referenced_object_id = cr.object_id AND fkc.referenced_column_id = cr.column_id ORDER BY tp.name;"
